# ============================================================================
# eCan.ai Linux Requirements
# ============================================================================
# Complete dependencies for running eCan.ai on Linux (Ubuntu/Debian/Fedora)
#
# Installation:
#   pip install -r requirements-linux.txt
#
# System dependencies (install first):
#   Ubuntu/Debian:
#     sudo apt-get install -y \
#       python3-dev python3-pip python3-venv \
#       gcc g++ make \
#       libffi-dev libssl-dev \
#       libgl1 libglib2.0-0 \
#       libxkbcommon-x11-0 libxcb-icccm4 libxcb-image0 \
#       libxcb-keysyms1 libxcb-randr0 libxcb-render-util0 \
#       libxcb-shape0 libxcb-xfixes0 libxcb-xinerama0 libxcb-cursor0 \
#       scrot wmctrl xdotool xdg-utils \
#       cups tesseract-ocr \
#       python3-tk
#
#   Fedora/RHEL:
#     sudo dnf install -y \
#       python3-devel gcc gcc-c++ make \
#       libffi-devel openssl-devel \
#       mesa-libGL glib2 \
#       libxkbcommon-x11 xcb-util-image xcb-util-keysyms \
#       xcb-util-renderutil xcb-util-wm \
#       scrot wmctrl xdotool xdg-utils \
#       cups tesseract \
#       python3-tkinter
# ============================================================================
